\section{Limitations}

\subsection{Simulation Simplifications}

\begin{itemize}
    \item \textbf{Perfect perception:} Agents have unlimited perception range, whereas real robots face sensor limitations and occlusion.
    \item \textbf{Instantaneous communication:} Messages are delivered without latency or packet loss, unlike real wireless networks.
    \item \textbf{Simplified movement:} Grid-based movement with unit speed does not capture continuous motion dynamics or kinematic constraints.
    \item \textbf{No obstacles:} The environment lacks obstacles, terrain variation, or dynamic hazards present in real agricultural settings.
\end{itemize}

\subsection{Agent Behavior Simplifications}

\begin{itemize}
    \item \textbf{Greedy movement:} Agents use simple greedy navigation rather than optimal path planning.
    \item \textbf{Broadcast communication:} All neighbors receive messages regardless of need, whereas selective communication might be more efficient.
    \item \textbf{No learning:} Agents do not adapt their strategies based on experience.
\end{itemize}

\subsection{Experimental Scope}

\begin{itemize}
    \item \textbf{Parameter ranges:} Communication range limited to 0--8 cells; team size limited to 10--30 agents.
    \item \textbf{Grid size:} Fixed 50×50 grid may not capture scalability effects in larger environments.
    \item \textbf{Single communication protocol:} Only broadcast protocol tested; peer-to-peer and stigmergic approaches not compared.
\end{itemize}

\subsection{Generalizability}

\begin{itemize}
    \item \textbf{Task specificity:} Findings are specific to fruit harvesting; other foraging tasks (e.g., prey capture, construction) may exhibit different dynamics.
    \item \textbf{Homogeneous agents:} All agents have identical capabilities; heterogeneous teams may require different communication strategies.
\end{itemize}

\section{Future Research Directions}

\subsection{Model Extensions}

\begin{itemize}
    \item \textbf{Adaptive communication:} Implement agents that dynamically adjust communication range based on local density or resource availability.
    \item \textbf{Reinforcement learning:} Train agents to learn optimal communication and coordination strategies through multi-agent RL.
    \item \textbf{Heterogeneous teams:} Introduce agents with varying capabilities (speed, perception, communication range) and study task allocation.
    \item \textbf{Realistic constraints:} Add sensor noise, communication latency, energy budgets, and kinematic constraints.
\end{itemize}

\subsection{Alternative Communication Protocols}

\begin{itemize}
    \item \textbf{Peer-to-peer messaging:} Compare targeted communication with broadcast.
    \item \textbf{Stigmergy:} Implement pheromone-based indirect communication.
    \item \textbf{Implicit communication:} Explore learned implicit coordination without explicit messages.
\end{itemize}

\subsection{Expanded Parameter Space}

\begin{itemize}
    \item \textbf{Larger teams:} Test scalability with 50--200 agents.
    \item \textbf{Larger environments:} Explore 100×100 to 500×500 grids.
    \item \textbf{Dynamic resources:} Implement time-varying fruit ripening patterns and spatial clustering.
\end{itemize}

\subsection{Real-World Validation}

\begin{itemize}
    \item \textbf{Hardware experiments:} Deploy communication protocols on physical robot teams in controlled environments.
    \item \textbf{Field trials:} Test in actual agricultural settings with real fruit harvesting tasks.
    \item \textbf{Human-robot teams:} Investigate communication between human supervisors and robot teams.
\end{itemize}
